cmake_minimum_required(VERSION 3.18)
project(stopwatch LANGUAGES C)
enable_testing()
add_executable(stopwatch_test portable_test.c)
target_compile_features(stopwatch_test PRIVATE c_std_11)
if(MSVC)
    target_compile_options(stopwatch_test PRIVATE /W3 /WX /UNDEBUG)
else()
    target_compile_options(stopwatch_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME stopwatch COMMAND stopwatch_test)
install(FILES stopwatch.h DESTINATION include)

if(NOT WIN32)
    find_package(Threads REQUIRED)
    add_executable(stopwatch_wall_test wall_test.c)
    target_compile_features(stopwatch_wall_test PRIVATE c_std_11)
    target_compile_options(stopwatch_wall_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(stopwatch_wall_test PRIVATE Threads::Threads)
    add_test(NAME stopwatch_wall COMMAND stopwatch_wall_test)
    add_executable(stopwatch_wall_slow_test wall_test.c)
    target_compile_features(stopwatch_wall_slow_test PRIVATE c_std_11)
    target_compile_options(stopwatch_wall_slow_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_compile_definitions(stopwatch_wall_slow_test PRIVATE STOPWATCH_WALL_MAX_SLEW_PPM=500)
    target_link_libraries(stopwatch_wall_slow_test PRIVATE Threads::Threads)
    add_test(NAME stopwatch_wall_slow COMMAND stopwatch_wall_slow_test)
endif()
